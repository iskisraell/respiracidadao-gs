def classify_threshold(metrica: str, valor: float) -> Tuple[Literal["normal", "atencao", "critico"], bool]:
    """Regra local: decide o nível pelos dois limiares, sem depender de rede."""
    low, high = THRESHOLDS[metrica]
    if valor >= high:
        return "critico", True
    if valor >= low:
        return "atencao", True
    return "normal", False
